The quote history page (`quotes.html`) provides a centralized interface for managing all your saved quotes. You can view past quotes, edit drafts, reissue quotes, and manage your quotation workflow.

## Accessing Quote History

Quote history requires authentication to access.

<Steps>
  <Step title="Sign In">
    Log in using your Firebase Authentication credentials from the main app.
  </Step>
  
  <Step title="Open Menu">
    Click the hamburger menu icon in the top-right corner.
  </Step>
  
  <Step title="Select History">
    Click "Ver historial" to open the quote history page.
  </Step>
</Steps>

```javascript
// Menu action to open history
if (historyAction) {
  historyAction.addEventListener('click', function() {
    closeMenu();
    if (!isAuthenticated()) {
      showToast('Inicia sesión para ver el historial.', 'error');
      openLoginModal();
      return;
    }
    window.location.href = 'quotes.html';
  });
}
```

<Warning>
  The "Ver historial" menu option is disabled when you're not authenticated. You must sign in first to access your quote history.
</Warning>

## Loading Quote History

Quotes are loaded automatically when the page opens.

### Bootstrap and Authentication Check

```javascript
async function bootstrap() {
  const user = window.FirebaseAuthService.getCurrentUser();
  const token = await window.FirebaseAuthService.getValidIdToken();
  
  if (!user || !token) {
    setAuthUi(false, '');
    return;
  }
  
  setAuthUi(true, user.email || '');
  try {
    await loadQuotes();
    showStatus('Historial actualizado.', false);
  } catch (e) {
    showStatus(e.message || 'No se pudo cargar historial.', true);
  }
}
```

### Fetching Quotes from Firebase

```javascript
async function loadQuotes() {
  const { user, idToken } = await requireAuthContext();
  const list = await window.QuotesRepo.listQuotes(user.uid, idToken);
  const container = document.getElementById('quoteList');
  
  if (!list.length) {
    container.innerHTML = 
      '<div class="quote-item">' +
      'No hay cotizaciones guardadas.' +
      '</div>';
    return;
  }
  
  container.innerHTML = list.map(quoteItemTemplate).join('');
}
```

<Info>
  Quotes are automatically sorted by update date, with the most recently modified quotes appearing first.
</Info>

## Quote Statuses

CotizaClima distinguishes between two quote statuses:

### Draft Status

Quotes saved without generating a PDF are marked as drafts.

```javascript
// Saving a draft quote
async function saveDraftQuote() {
  return persistCurrentQuote('draft', { 
    requireAuth: true, 
    silent: false 
  });
}
```

<Note>
  Draft quotes are useful for:
  - Incomplete quotes that need more information
  - Quotes pending approval from a supervisor
  - Templates for similar future quotes
</Note>

### Issued Status

Quotes are marked as issued when you generate a PDF.

```javascript
// Auto-save as issued when generating PDF
window.generatePDF = async function patchedGeneratePDF() {
  let saved = null;
  try {
    saved = await persistCurrentQuote('issued', { 
      requireAuth: false, 
      silent: true 
    });
  } catch (e) {
    saved = null;
  }
  
  await original(); // Generate the PDF
  
  if (saved) {
    showToast('PDF generado y cotización guardada', 'success');
  }
};
```

<Info>
  Issued quotes represent final versions that have been provided to clients. You can still edit and reissue them if needed.
</Info>

### Duplicate Quote

Create a copy of an existing quote with a new quote number.

```javascript
async function duplicateQuote(id) {
  const { user, idToken } = await requireAuthContext();
  const quote = await window.QuotesRepo.getQuote(user.uid, idToken, id);
  if (!quote) throw new Error('No se encontró para duplicar.');
  
  // Calculate next quote number for the same year
  const targetYear = getQuoteYearFromNumber(quote.quoteNumber);
  const allQuotes = await window.QuotesRepo.listQuotes(user.uid, idToken);
  const maxSeq = (allQuotes || []).reduce((max, item) => {
    const seq = parseQuoteSequenceByYear(
      item && item.quoteNumber, 
      targetYear
    );
    return seq > max ? seq : max;
  }, 0);
  const nextQuoteNumber = buildQuoteNumber(targetYear, maxSeq + 1);
  
  // Create clone with new ID and quote number
  const clone = {
    ...quote,
    id: undefined,
    activeQuoteId: null,
    status: 'draft',
    quoteNumber: nextQuoteNumber,
    createdAt: Date.now(),
    updatedAt: Date.now()
  };
  
  await window.QuotesRepo.saveQuote(user.uid, idToken, clone, null);
  showStatus('Cotización duplicada.', false);
  await loadQuotes();
}
```

<Info>
  Duplicating is useful for:
  - Creating similar quotes for different clients
  - Using a past quote as a template
  - Reissuing an old quote with updated pricing
</Info>

### Delete Quote

Permanently remove a quote from your history.

```javascript
async function deleteQuote(id) {
  const ok = window.confirm(
    '¿Eliminar esta cotización? Esta acción no se puede deshacer.'
  );
  if (!ok) return;
  
  const { user, idToken } = await requireAuthContext();
  await window.QuotesRepo.deleteQuote(user.uid, idToken, id);
  showStatus('Cotización eliminada.', false);
  await loadQuotes();
}
```

<Warning>
  Deletion is permanent and cannot be undone. A confirmation dialog will appear before deleting.
</Warning>

### Navigate with Waze

Open Waze with the client's address for turn-by-turn navigation.

```javascript
function buildWazeUrl(client) {
  if (!client) return '';
  const parts = [client.address, client.city, client.region]
    .filter(Boolean);
  if (!parts.length) return '';
  return 'https://waze.com/ul?q=' + 
         encodeURIComponent(parts.join(', ')) + 
         '&navigate=yes';
}
```

<Tip>
  The Waze button is especially useful when you need to visit a client's location for installation or service.
</Tip>

## Quote Number Format

CotizaClima uses a structured quote numbering system.

### Current Format

Format: `CL-YYYY-####`

- `CL` = Company prefix (CotizaClima)
- `YYYY` = Four-digit year
- `####` = Sequential number (zero-padded to 4 digits)

Examples:
- `CL-2026-0001` (First quote of 2026)
- `CL-2026-0042` (42nd quote of 2026)
- `CL-2025-0999` (999th quote of 2025)

### Legacy Format Support

Older quotes may use the legacy format:

Format: `COT-YYYYMMDD-####`

```javascript
function parseQuoteSequenceByYear(quoteNumber, year) {
  const normalized = String(quoteNumber || '').trim().toUpperCase();
  
  // Current format: CL-YYYY-####
  const currentFmt = normalized.match(/^CL-(\d{4})-(\d{4,})$/);
  if (currentFmt) {
    if (String(year) !== currentFmt[1]) return 0;
    return Number(currentFmt[2] || 0) || 0;
  }
  
  // Legacy format: COT-YYYYMMDD-####
  const legacyFmt = normalized.match(/^COT-(\d{8})-(\d{4,})$/);
  if (legacyFmt) {
    const legacyYear = legacyFmt[1].slice(0, 4);
    if (String(year) !== legacyYear) return 0;
    return Number(legacyFmt[2] || 0) || 0;
  }
  
  return 0;
}
```

<Info>
  CotizaClima automatically handles both current and legacy quote number formats for backward compatibility.
</Info>

## Quote Repository Operations

All quote operations go through the `QuotesRepo` module.

### List All Quotes

```javascript
async function listQuotes(uid, idToken) {
  const raw = await request(`quotes/${uid}`, idToken, { 
    method: 'GET' 
  });
  if (!raw || typeof raw !== 'object') return [];
  
  return Object.entries(raw)
    .map(([id, payload]) => ({ id, ...(payload || {}) }))
    .sort((a, b) => 
      Number(b.updatedAt || b.createdAt || 0) - 
      Number(a.updatedAt || a.createdAt || 0)
    );
}
```

### Get Single Quote

```javascript
async function getQuote(uid, idToken, quoteId) {
  const payload = await request(
    `quotes/${uid}/${quoteId}`, 
    idToken, 
    { method: 'GET' }
  );
  if (!payload) return null;
  return { id: quoteId, ...payload };
}
```

### Save or Update Quote

```javascript
async function saveQuote(uid, idToken, quotePayload, quoteId) {
  const now = Date.now();
  const basePayload = {
    ...quotePayload,
    ownerUid: uid,
    updatedAt: now,
    createdAt: quotePayload && quotePayload.createdAt 
      ? quotePayload.createdAt 
      : now
  };
  
  if (quoteId) {
    // Update existing quote
    await request(`quotes/${uid}/${quoteId}`, idToken, {
      method: 'PUT',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(basePayload)
    });
    return { id: quoteId, payload: basePayload };
  }
  
  // Create new quote
  const created = await request(`quotes/${uid}`, idToken, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(basePayload)
  });
  
  const id = created && created.name ? created.name : null;
  if (!id) throw new Error('No se pudo crear la cotización (sin id).');
  return { id, payload: basePayload };
}
```

### Delete Quote

```javascript
async function deleteQuote(uid, idToken, quoteId) {
  await request(`quotes/${uid}/${quoteId}`, idToken, { 
    method: 'DELETE' 
  });
  return true;
}
```

## Troubleshooting

<Accordion title="Quotes not loading">
  **Possible causes:**
  - Lost authentication session
  - Firebase connection issues
  - Browser offline
  
  **Solution:** Sign out and sign in again, check your internet connection, and try refreshing the page.
</Accordion>

<Accordion title="Cannot open a quote for editing">
  **Possible cause:** Quote data is corrupted or incomplete
  
  **Solution:** Try duplicating the quote instead, which creates a fresh copy.
</Accordion>

<Accordion title="Delete confirmation not appearing">
  **Possible cause:** Browser is blocking JavaScript dialogs
  
  **Solution:** Allow dialogs for the CotizaClima site in your browser settings.
</Accordion>

<Note>
  Quote history is stored per user account. Different accounts will have completely separate quote histories.
</Note>
