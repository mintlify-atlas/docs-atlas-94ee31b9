CotizaClima generates professional PDF quotations using jsPDF and jsPDF-AutoTable. The generated PDFs include complete quotation details, service descriptions, pricing breakdowns, and optional company branding.

## PDF Library Integration

The application uses two jsPDF libraries:

```html title="index.html"
<script src="https://cdnjs.cloudflare.com/ajax/libs/jspdf/2.5.1/jspdf.umd.min.js"></script>
<script src="https://cdnjs.cloudflare.com/ajax/libs/jspdf-autotable/3.6.0/jspdf.plugin.autotable.min.js"></script>
```

- **jsPDF**: Core PDF generation library
- **jsPDF-AutoTable**: Plugin for creating tables with automatic pagination

## PDF Generation Flow

```javascript {2-5,8-10,13-15} title="app.js"
async function generatePDF() {
  const { jsPDF } = window.jspdf;
  const doc = new jsPDF({ 
    orientation: 'portrait', 
    unit: 'mm', 
    format: 'a4' 
  });
  
  const pageW = 210, margin = 18;
  let y = 0;
  
  // 1. Header with logo and company info
  y = await renderPdfHeader(doc, pageW, margin);
  
  // 2. Client information section
  y = renderClientSection(doc, y, margin, pageW);
  
  // 3. Equipment and services table
  y = await renderItemsTable(doc, y, margin, pageW);
  
  // 4. Pricing totals
  y = renderTotals(doc, y, pageW, margin);
  
  // 5. Service notes
  y = renderNotes(doc, y, margin, pageW);
  
  // 6. Service-specific information
  y = appendServiceInfoToPdf(doc, y, margin, pageW, selectedServiceItems);
  
  // 7. Footer
  renderFooter(doc, pageW);
  
  doc.save(`${quoteNumber}.pdf`);
  showToast('PDF generado correctamente', 'success');
}
```

```javascript title="app.js"
  doc.setFillColor(31, 53, 82);

  doc.setTextColor(255, 255, 255);
  doc.setFontSize(8);
  doc.setFont('helvetica', 'bold');
```

## Equipment and Services Table

The main table uses jsPDF-AutoTable for automatic pagination:

```javascript {5-15,18-35,40-48} title="app.js"
const selectedEquipItems = getSelectedEquipmentItems();
const selectedServiceItems = getSelectedServiceItems();
const activeExtras = ACCESSORIES.filter(acc => (accQty[acc.id]||0) > 0);

const tableBody = [];

// Add equipment rows
selectedEquipItems.forEach(item => {
  tableBody.push([
    `${item.ac.brand_model}\n${item.ac.type} · ${item.ac.refrigerant} · Kit ${item.ac.installation_kit}`,
    `${(item.btu/1000).toFixed(0)}.000 BTU`,
    String(item.qty),
    `$${fmtNum(item.cap.net_price)}`,
    `$${fmtNum(item.cap.price_with_tax * item.qty)}`
  ]);
});

// Add service rows
selectedServiceItems.forEach(item => {
  const serviceDesc = `${item.svc.name}\n${item.svc.desc} · ${item.qty} x $${fmtNum(item.unitPrice)}`;
  tableBody.push([
    serviceDesc,
    'Servicio',
    String(item.qty),
    item.totalPrice > 0 ? `$${fmtNum(Math.round(item.totalPrice/1.19))}` : '$0',
    item.totalPrice > 0 ? `$${fmtNum(item.totalPrice)}` : 'Sin costo'
  ]);
});

// Add accessory rows
activeExtras.forEach(acc => {
  const q = accQty[acc.id];
  tableBody.push([
    `${acc.name}\n${acc.desc}`,
    'Accesorio',
    String(q),
    `$${fmtNum(acc.price)}`,
    `$${fmtNum(acc.price * q)}`
  ]);
});

// Render table
doc.autoTable({
  startY: y,
  head: [['DESCRIPCIÓN', 'TIPO', 'CANT.', 'P. NETO', 'P. c/IVA']],
  body: tableBody,
  margin: { left: margin, right: margin },
  styles: { fontSize: 8, font: 'helvetica', cellPadding: 4 },
  headStyles: {
    fillColor: [31, 53, 82],
    textColor: [255, 255, 255],
    fontStyle: 'bold'
  },
  alternateRowStyles: { fillColor: [247, 248, 250] },
  columnStyles: {
    0: { cellWidth: 82 },
    1: { cellWidth: 22, halign: 'center' },
    2: { cellWidth: 14, halign: 'center' },
    3: { cellWidth: 28, halign: 'right' },
    4: { cellWidth: 28, halign: 'right', fontStyle: 'bold' }
  },
});

y = doc.lastAutoTable.finalY + 8;
```

<Note>
  The `\n` character creates multi-line cells. AutoTable automatically wraps long text.
</Note>

```html title="index.html"
             min="0" max="100" 
             oninput="recalcTotal()">
    </div>
    <div class="summary-row-val discount-badge" id="discountBadgeAmt">— $0</div>
  </div>
</div>
```

The discount is recalculated on input:

```javascript {5-7} title="app.js"
function recalcTotal() {
  const equipPrice = parseFloat(document.getElementById('sumEquipPrice').value) || 0;
  const servicePrice = parseFloat(document.getElementById('sumServicePrice').value) || 0;
  let extrasTotal = 0;
  ACCESSORIES.forEach(acc => {
    extrasTotal += (accQty[acc.id]||0) * acc.price;
  });
  
  const subtotal = equipPrice + servicePrice + extrasTotal;
  const discPct = parseFloat(document.getElementById('discountPct').value) || 0;
  const discAmt = subtotal * discPct / 100;
  const total = subtotal - discAmt;
  
  document.getElementById('discountBadgeAmt').textContent = `−$${fmtNum(Math.round(discAmt))}`;
  document.getElementById('grandTotal').textContent = fmtNum(Math.round(total));
}
```

## Service-Specific Content

Each service type includes detailed information in the PDF:

```javascript {7-11,15-29} title="app.js"
function appendServiceInfoToPdf(doc, startY, margin, pageW, selectedServiceItems) {
  const keys = getServiceInfoKeys(selectedServiceItems);
  if (!keys.length) return startY;
  
  let y = startY;
  
  keys.forEach((key, idx) => {
    const block = SERVICE_PDF_CONTENT[key];
    if (!block) return;
    
    y = ensurePdfSpace(doc, y, 38);
    sectionHeader(doc, `  INFORMACION RELEVANTE · ${block.title.toUpperCase()}`, margin, y, pageW - margin*2);
    y += 8;
    
    // Relevant information
    doc.setFont('helvetica', 'bold');
    doc.text('Información relevante:', margin, y);
    y += 4;
    
    doc.setFont('helvetica', 'normal');
    block.relevant.forEach(paragraph => {
      const lines = doc.splitTextToSize(paragraph, pageW - margin*2);
      y = ensurePdfSpace(doc, y, lines.length * 4 + 2);
      doc.text(lines, margin, y);
      y += lines.length * 4 + 1;
    });
    
    // Extra information (bullet points)
    y += 2;
    doc.setFont('helvetica', 'bold');
    doc.text('Información extra:', margin, y);
    y += 4;
    
    doc.setFont('helvetica', 'normal');
    block.extra.forEach(item => {
      const lines = doc.splitTextToSize(`- ${item}`, pageW - margin*2);
      y = ensurePdfSpace(doc, y, lines.length * 4 + 1);
      doc.text(lines, margin, y);
      y += lines.length * 4 + 1;
    });
  });
  
  return y;
}
```

<Tip>
  The `ensurePdfSpace` function automatically adds a new page if content would overflow the current page.
</Tip>

### Automatic Pagination

```javascript title="app.js"
function ensurePdfSpace(doc, y, neededHeight, pageH = 297) {
  if (y + neededHeight <= pageH - 22) return y;
  doc.addPage();
  return 20; // Start new page with top margin
}
```

## Auto-Save on PDF Generation

Quotations are automatically saved when generating PDF:

```javascript {5-9} title="quote-persistence.js"
function patchGeneratePdfForAutosave() {
  const original = window.generatePDF;
  
  window.generatePDF = async function patchedGeneratePDF() {
    let saved = null;
    try {
      saved = await persistCurrentQuote('issued', { requireAuth: false, silent: true });
    } catch (e) {
      saved = null;
    }
    
    await original();
    
    if (saved) {
      showToast('PDF generado y cotización guardada', 'success');
    }
  };
}
```

<Info>
  The auto-save is silent and doesn't interrupt the PDF download even if Firebase is unavailable.
</Info>
